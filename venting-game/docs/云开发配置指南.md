# 云开发配置指南

## 一、云开发控制台配置

### 1. 开通云开发
1. 打开微信开发者工具
2. 点击顶部菜单 **云开发** → **开通**
3. 创建环境，填写环境名称（如：venting-game-dev）
4. 记录下 **环境ID**（格式如：cloud1-xxxxxxxxxxxx）

### 2. 创建数据库集合

在云开发控制台 → 数据库，创建以下集合：

#### users 集合（用户信息）
```json
{
  "_id": "自动生成",
  "_openid": "用户openid（自动添加）",
  "nickname": "用户昵称",
  "avatarUrl": "头像URL",
  "createdAt": {"$date": "创建时间"},
  "updatedAt": {"$date": "更新时间"}
}
```
**权限设置**：所有用户可读，仅创建者可写

#### targets 集合（发泄目标）
```json
{
  "_id": "自动生成",
  "_openid": "用户openid（自动添加）",
  "name": "目标名称",
  "type": "cartoon/photo",
  "imageUrl": "图片URL",
  "cartoonId": "卡通ID（如果是卡通）",
  "gender": "male/female",
  "createdAt": {"$date": "创建时间"},
  "lifetimeStats": {
    "totalAttacks": 0,
    "attacks": {
      "slap": 10,
      "punch": 5
    }
  },
  "progress": {
    "level": 1,
    "exp": 0
  }
}
```
**权限设置**：所有用户可读，仅创建者可写

#### attack_records 集合（攻击记录）
```json
{
  "_id": "自动生成",
  "_openid": "用户openid（自动添加）",
  "targetId": "目标ID",
  "skillId": "技能ID",
  "damage": "伤害值",
  "isCrit": "是否暴击",
  "createdAt": {"$date": "攻击时间"}
}
```
**权限设置**：所有用户可读，仅创建者可写

### 3. 数据库权限配置

在每个集合的 **权限设置** 中选择：
- **自定义权限**
```json
{
  "read": true,
  "write": "doc._openid == auth.openid"
}
```

## 二、云函数上传部署

### 1. 上传 login 云函数
1. 右键点击 `cloudfunctions/login` 文件夹
2. 选择 **上传并部署：云端安装依赖**
3. 等待部署完成

### 2. 上传 removeBackground 云函数（如果需要抠图功能）
1. 右键点击 `cloudfunctions/removeBackground` 文件夹
2. 选择 **上传并部署：云端安装依赖**
3. 等待部署完成

### 3. 配置云函数环境变量（可选）

如果需要使用腾讯云抠图API，在 `removeBackground` 云函数中配置：
- 云开发控制台 → 云函数 → removeBackground → 配置
- 添加环境变量：
  - `TENCENT_SECRET_ID`: 你的腾讯云密钥ID
  - `TENCENT_SECRET_KEY`: 你的腾讯云密钥Key

### 2. 添加登录功能

创建 `utils/cloud.js` 用于封装云开发操作：

```javascript
// utils/cloud.js

/**
 * 获取用户 openid
 */
const getOpenId = async () => {
  try {
    const res = await wx.cloud.callFunction({
      name: 'login'
    })
    return res.result.openid
  } catch (err) {
    console.error('获取openid失败:', err)
    return null
  }
}

/**
 * 获取用户信息
 */
const getUserInfo = async () => {
  try {
    const db = wx.cloud.database()
    const res = await db.collection('users').get()
    if (res.data.length > 0) {
      return res.data[0]
    }
    return null
  } catch (err) {
    console.error('获取用户信息失败:', err)
    return null
  }
}

/**
 * 创建/更新用户信息
 */
const saveUserInfo = async (userInfo) => {
  try {
    const db = wx.cloud.database()
    const _ = db.command

    // 检查用户是否存在
    const existing = await getUserInfo()
    const now = new Date()

    if (existing) {
      // 更新用户
      const res = await db.collection('users').doc(existing._id).update({
        data: {
          ...userInfo,
          updatedAt: db.serverDate()
        }
      })
      return { ...existing, ...userInfo }
    } else {
      // 创建用户
      const res = await db.collection('users').add({
        data: {
          ...userInfo,
          createdAt: db.serverDate(),
          updatedAt: db.serverDate()
        }
      })
      return { _id: res._id, ...userInfo }
    }
  } catch (err) {
    console.error('保存用户信息失败:', err)
    throw err
  }
}

/**
 * 获取用户的所有发泄目标
 */
const getMyTargets = async () => {
  try {
    const db = wx.cloud.database()
    const res = await db.collection('targets').orderBy('createdAt', 'desc').get()
    return res.data
  } catch (err) {
    console.error('获取发泄目标失败:', err)
    return []
  }
}

/**
 * 创建发泄目标
 */
const createTarget = async (target) => {
  try {
    const db = wx.cloud.database()
    const res = await db.collection('targets').add({
      data: {
        ...target,
        createdAt: db.serverDate()
      }
    })
    return { _id: res._id, ...target }
  } catch (err) {
    console.error('创建发泄目标失败:', err)
    throw err
  }
}

/**
 * 更新发泄目标
 */
const updateTarget = async (targetId, updateData) => {
  try {
    const db = wx.cloud.database()
    await db.collection('targets').doc(targetId).update({
      data: updateData
    })
    return true
  } catch (err) {
    console.error('更新发泄目标失败:', err)
    throw err
  }
}

/**
 * 删除发泄目标
 */
const deleteTarget = async (targetId) => {
  try {
    const db = wx.cloud.database()
    await db.collection('targets').doc(targetId).remove()
    return true
  } catch (err) {
    console.error('删除发泄目标失败:', err)
    throw err
  }
}

/**
 * 记录攻击
 */
const recordAttack = async (attackData) => {
  try {
    const db = wx.cloud.database()
    const res = await db.collection('attack_records').add({
      data: {
        ...attackData,
        createdAt: db.serverDate()
      }
    })
    return res._id
  } catch (err) {
    console.error('记录攻击失败:', err)
    throw err
  }
}

module.exports = {
  getOpenId,
  getUserInfo,
  saveUserInfo,
  getMyTargets,
  createTarget,
  updateTarget,
  deleteTarget,
  recordAttack
}
```

## 四、修改页面代码使用云数据库

### 1. 修改 app.js 添加登录逻辑

```javascript
// app.js
import { getOpenId, getUserInfo, saveUserInfo } from './utils/cloud.js'

App({
  data: {
    openid: null,
    userInfo: null
  },

  onLaunch() {
    this.initLogin()
  },

  async initLogin() {
    try {
      // 获取 openid
      const openid = await getOpenId()
      if (!openid) {
        console.error('获取openid失败')
        return
      }

      this.setData({ openid })

      // 获取用户信息
      let userInfo = await getUserInfo()

      if (!userInfo) {
        // 首次登录，创建用户
        const systemInfo = wx.getSystemInfoSync()
        userInfo = await saveUserInfo({
          nickname: '小卡拉' + Math.floor(Math.random() * 10000),
          avatarUrl: ''
        })
      }

      this.setData({ userInfo })
      console.log('登录成功:', userInfo)

    } catch (err) {
      console.error('登录失败:', err)
    }
  }
})
```

### 2. 修改 index.js 使用云数据库

```javascript
// pages/index/index.js
import { getMyTargets, deleteTarget } from '../../utils/cloud.js'

Page({
  data: {
    targets: []
  },

  onLoad() {
    this.loadTargets()
  },

  async loadTargets() {
    wx.showLoading({ title: '加载中...' })
    try {
      const targets = await getMyTargets()
      this.setData({ targets })
    } catch (err) {
      wx.showToast({
        title: '加载失败',
        icon: 'none'
      })
    } finally {
      wx.hideLoading()
    }
  },

  async deleteTarget(e) {
    const id = e.currentTarget.dataset.id
    try {
      await deleteTarget(id)
      wx.showToast({ title: '删除成功' })
      this.loadTargets()
    } catch (err) {
      wx.showToast({ title: '删除失败', icon: 'none' })
    }
  }
})
```

## 六、常见问题

### Q1: 云函数调用失败
- 检查环境ID是否正确
- 检查云函数是否已上传部署
- 检查网络连接

### Q2: 数据库权限错误
- 检查集合权限设置
- 确保使用 `doc._openid == auth.openid` 规则

### Q3: openid 为 undefined
- 确保 app.json 中配置了 `"cloud": true`
- 确保云开发环境已开通
- 重新编译小程序

### Q4: 数据保存失败
- 检查数据格式是否符合数据库结构
- 查看控制台错误信息
- 检查云函数日志
