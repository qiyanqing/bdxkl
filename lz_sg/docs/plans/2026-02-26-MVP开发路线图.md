### 阶段3：战斗逻辑核心（2小时）

#### 3.1 核心战斗流程
```javascript
// utils/combat.js（简化版核心逻辑）

class Combat {
  constructor(myHeroes, enemyHeroes) {
    this.myHeroes = myHeroes;      // 我方6个角色
    this.enemyHeroes = enemyHeroes; // 敌方6个角色
    this.turnQueue = [];           // 行动队列
    this.currentTurn = 0;
  }

  // 初始化战斗
  init() {
    // 合并双方角色，按速度排序
    const allHeroes = [...this.myHeroes, ...this.enemyHeroes];
    this.turnQueue = allHeroes.sort((a, b) => b.speed - a.speed);
  }

  // 执行一回合
  async executeTurn() {
    const hero = this.turnQueue[this.currentTurn];

    // 判断是否存活
    if (hero.currentHp <= 0) {
      this.currentTurn++;
      return;
    }

    // 选择目标（AI逻辑）
    const target = this.selectTarget(hero);

    // 执行攻击
    await this.attack(hero, target);

    // 回蓝
    hero.currentMp = Math.min(hero.maxMp, hero.currentMp + hero.mpRecovery);

    // 检查蓝条是否满，满则释放大招
    if (hero.currentMp >= hero.maxMp) {
      await this.useSkill(hero, target);
    }

    // 检查战斗结束
    if (this.checkBattleEnd()) {
      return this.getBattleResult();
    }

    // 下一回合
    this.currentTurn = (this.currentTurn + 1) % this.turnQueue.length;
  }

  // 选择目标（简化AI）
  selectTarget(attacker) {
    const isMyTeam = this.myHeroes.includes(attacker);
    const enemies = isMyTeam ? this.enemyHeroes : this.myHeroes;

    // 找血量最少的敌人
    return enemies.reduce((min, enemy) =>
      enemy.currentHp < min.currentHp ? enemy : min
    );
  }

  // 普通攻击
  async attack(attacker, target) {
    const damage = Math.max(1, attacker.atk - target.def);
    target.currentHp -= damage;

    // 触发UI更新（通过回调）
    if (this.onAttack) {
      await this.onAttack(attacker, target, damage);
    }
  }

  // 释放技能
  async useSkill(hero, target) {
    const damage = Math.floor(hero.atk * hero.skill.damage - target.def);
    target.currentHp -= damage;
    hero.currentMp = 0; // 消耗全部蓝量

    // 触发UI更新
    if (this.onSkill) {
      await this.onSkill(hero, target, damage);
    }
  }

  // 检查战斗结束
  checkBattleEnd() {
    const myAlive = this.myHeroes.filter(h => h.currentHp > 0).length;
    const enemyAlive = this.enemyHeroes.filter(h => h.currentHp > 0).length;

    return myAlive === 0 || enemyAlive === 0;
  }

  // 获取战斗结果
  getBattleResult() {
    const myAlive = this.myHeroes.filter(h => h.currentHp > 0).length;
    return myAlive > 0 ? 'win' : 'lose';
  }
}

module.exports = Combat;
```

#### 4.3 页面逻辑
```javascript
// pages/battle/battle.js
const Combat = require('../../utils/combat.js');
const { heroes } = require('../../data/characters.js');

Page({
  data: {
    myHeroes: [],
    enemyHeroes: [],
    turn: 0,
    currentAction: ''
  },

  onLoad() {
    // 初始化我方角色
    const myHeroes = heroes.map(h => ({
      ...h,
      currentHp: h.maxHp,
      currentMp: 0
    }));

    // 初始化敌方角色（复制我方数据作为测试）
    const enemyHeroes = heroes.map(h => ({
      ...h,
      id: h.id + '_enemy',
      name: h.name + '（敌）',
      currentHp: h.maxHp,
      currentMp: 0
    }));

    this.setData({ myHeroes, enemyHeroes });

    // 创建战斗实例
    this.combat = new Combat(myHeroes, enemyHeroes);

    // 绑定UI回调
    this.combat.onAttack = this.onAttack.bind(this);
    this.combat.onSkill = this.onSkill.bind(this);

    // 初始化战斗
    this.combat.init();

    // 开始战斗循环
    this.startBattle();
  },

  async startBattle() {
    const result = await this.combat.executeTurn();

    if (result) {
      // 战斗结束
      this.setData({ currentAction: `战斗${result === 'win' ? '胜利' : '失败'}！` });
      return;
    }

    // 更新UI
    this.setData({
      myHeroes: this.combat.myHeroes,
      enemyHeroes: this.combat.enemyHeroes,
      turn: this.data.turn + 1
    });

    // 延迟后继续下一回合
    setTimeout(() => {
      this.startBattle();
    }, 1000);
  },

  async onAttack(attacker, target, damage) {
    this.setData({ currentAction: `${attacker.name} 攻击 ${target.name}，造成 ${damage} 点伤害` });
    await this.delay(500);
  },

  async onSkill(hero, target, damage) {
    this.setData({ currentAction: `${hero.name} 释放 ${hero.skill.name}！造成 ${damage} 点伤害` });
    await this.delay(1000);
  },

  delay(ms) {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
});
```

## 七、常见问题

**Q: 战斗太快看不清？**
A: 在 `startBattle()` 中增加 `setTimeout` 延迟时间

**Q: 想暂停战斗？**
A: 添加暂停标志位，在战斗循环中检查

**Q: 想手动指定攻击目标？**
A: 点击卡牌时设置 `target`，传递给 `combat.executeTurn()`

**Q: 想重置战斗？**
A: 添加重置按钮，重新初始化战斗数据
